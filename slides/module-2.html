02 · Первая нормальная форма (1НФ)

1НФ: один карман — одна вещь (атомарность)

✗ Было (нарушение 1НФ)

101 Алиса Ким Худи SQL, Стикерпак
102 Тимур Омаров Термокружка

В ячейке items_list — текстовая каша через запятую. Поиск LIKE '%...%' медленный, количество штук и цену по товару не посчитать.

✓ Стало (1НФ: 1 товар = 1 строка)

101Стикерпак1

В каждой ячейке лежит одно неделимое значение, у каждой строки есть PRIMARY KEY — легко считать суммы и фильтровать.

🎒
Аналогия рюкзака: один карман — одна вещь. Никаких списков через запятую и никаких повторяющихся колонок item_1, item_2, item_3!
02 · Вторая нормальная форма (2НФ)

2НФ: каждый факт зависит от ВСЕГО ключа

Когда у таблицы составной ключ (order_id, prod_id), неключевые колонки не имеют права зависеть только от одной его половинки:

✗ Было (нарушение 2НФ)

3 2 1 Стикерпак
1 1 1 Худи «SQL Master»

  • title зависит только от половинки ключа prod_id и дублируется в каждом заказе.

02 · Практика 3НФ

Собираем чистую схему и проверяем победу над аномалиями

SQL 
DROP TABLE IF EXISTS product_tags, tags, shop_order_items, shop_orders, shop_products, shop_customers, cities CASCADE;

CREATE TABLE cities (
    id       INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name     VARCHAR(100) NOT NULL UNIQUE,
    timezone VARCHAR(20)  NOT NULL
);
CREATE TABLE shop_customers (
    id        INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    full_name VARCHAR(120) NOT NULL,
    phone     VARCHAR(30)  NOT NULL UNIQUE,
    city_id   INT NOT NULL REFERENCES cities(id)
);
CREATE TABLE shop_products (
    id    INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    title VARCHAR(150)   NOT NULL,
    price NUMERIC(10, 2) NOT NULL CHECK (price >= 0)
);
CREATE TABLE shop_orders (
    id          INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    customer_id INT NOT NULL REFERENCES shop_customers(id),
    created_at  TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE shop_order_items (
    order_id   INT NOT NULL REFERENCES shop_orders(id) ON DELETE CASCADE,
    product_id INT NOT NULL REFERENCES shop_products(id),
    quantity   INT NOT NULL CHECK (quantity > 0),
    unit_price NUMERIC(10, 2) NOT NULL CHECK (unit_price >= 0),
    PRIMARY KEY (order_id, product_id)
);
-- Наполняем 5 таблиц (полный INSERT — в кнопке «SQL урока 2.1») и собираем чек:
SELECT o.id AS order_id, c.full_name AS customer, ci.name AS city,
       p.title AS product, oi.quantity, oi.unit_price
FROM shop_orders o
JOIN shop_customers   c  ON c.id  = o.customer_id
JOIN cities           ci ON ci.id = c.city_id
JOIN shop_order_items oi ON oi.order_id = o.id
JOIN shop_products    p  ON p.id  = oi.product_id
ORDER BY o.id, p.id;
-- 1. Нет аномалии обновления: меняем телефон Алисы ОДИН раз
UPDATE shop_customers SET phone = '+7-701-555-00-00' WHERE id = 1;

-- 2. Нет аномалии вставки: добавляем товар без единого заказа
INSERT INTO shop_products (title, price) VALUES ('Блокнот «ACID»', 2500.00);

-- 3. Проверяем заказы: новый телефон сам подтянулся везде!
SELECT o.id AS order_id, c.full_name, c.phone
FROM shop_orders o JOIN shop_customers c ON c.id = o.customer_id
ORDER BY o.id;

Сначала создаём независимые таблицы (cities, shop_products), затем shop_customers со ссылкой на cities.

CREATE TABLE product_tags (
    product_id INT NOT NULL REFERENCES shop_products(id) ON DELETE CASCADE,
    tag_id     INT NOT NULL REFERENCES tags(id)          ON DELETE CASCADE,
    PRIMARY KEY (product_id, tag_id)

04 · Классическая ловушка миграций

Добавляем NOT NULL в непустую таблицу: 3 шага

SQL Пример 6 · Наивная попытка добавить обязательный артикул sku 
ALTER TABLE shop_products
ADD COLUMN sku VARCHAR(30) NOT NULL;
Ошибка PostgreSQL · В таблице уже 5 строк, а значение по умолчанию не задано!
ERROR:  column "sku" of relation "shop_products" contains null values
✗ Как НЕ надо
  • Добавлять NOT NULL без DEFAULT в заполненную таблицу — запрос сразу падает.
  • Запускать скрипт из 5 команд без BEGIN ... COMMIT: на 3-й команде база застрянет наполовину переделанной.
✓ Как правильно (3 шага в транзакции)
  1. Добавить колонку пока без NOT NULL (nullable).
  2. Заполнить старые строки (UPDATE — backfill).
  3. Затянуть гайки: SET NOT NULL + UNIQUE и COMMIT.
04 · Практика миграций

UP-миграция 001, откат DOWN и запуск в Docker

SQL 
CREATE TABLE IF NOT EXISTS schema_migrations (
    version VARCHAR(100) PRIMARY KEY, applied_at TIMESTAMP NOT NULL DEFAULT NOW()
);

BEGIN;
ALTER TABLE shop_products ADD COLUMN sku VARCHAR(30);
UPDATE shop_products SET sku = 'MERCH-' || LPAD(id::TEXT, 4, '0') WHERE sku IS NULL;
ALTER TABLE shop_products
    ALTER COLUMN sku SET NOT NULL,
    ADD CONSTRAINT shop_products_sku_unique UNIQUE (sku);
INSERT INTO schema_migrations (version) VALUES ('001_add_product_sku');
COMMIT;
-- Файл отката 001_add_product_sku.down.sql (для справки; не запускай, если хочешь оставить sku!)
BEGIN;

ALTER TABLE shop_products
    DROP CONSTRAINT IF EXISTS shop_products_sku_unique,
    DROP COLUMN IF EXISTS sku;

DELETE FROM schema_migrations
WHERE version = '001_add_product_sku';

COMMIT;
# Применяем файл миграции к сервису pg_primary (пользователь student, БД learning_db)
# ON_ERROR_STOP=1 немедленно остановит psql при любой ошибке
docker compose exec -T pg_primary psql -U student -d learning_db \
  -v ON_ERROR_STOP=1 < migrations/001_add_product_sku.up.sql

02 · Главная ловушка LEFT JOIN

Фильтр правой таблицы: WHERE против ON

Задача: «Покажи всех клиентов и только их оплаченные (paid) заказы. Если оплаченных нет — клиент всё равно должен остаться!»

✗ Как НЕ надо (фильтр в WHERE)
SELECT c.full_name, o.id
FROM customers AS c
LEFT JOIN orders AS o
  ON o.customer_id = c.id
WHERE o.status = 'paid';
  • WHERE срабатывает после склейки таблиц.
  • Вера (cancelled) и Глеб (NULL) не проходят проверку и исчезают.
  • LEFT JOIN тайно превратился в INNER JOIN!
✓ Как правильно (фильтр в ON)
SELECT c.full_name, o.id
FROM customers AS c
LEFT JOIN orders AS o
  ON o.customer_id = c.id
 AND o.status = 'paid';
  • Условие в ON отсеивает отменённые заказы до присоединения.
  • Алиса, Борис и Дина получают номера оплаченных заказов.
  • Вера и Глеб честно остаются в списке с NULL!
⚠️
Железное правило: любое условие на колонку правой таблицы внутри WHERE (кроме IS NULL) уничтожает LEFT JOIN. Переноси такие фильтры в блок ON!
Мини-квиз #1 · Проверка понимания

Кто останется в отчёте после WHERE o.status = 'paid'?

05 · Агрегация данных

GROUP BY, HAVING и полный конвейер выполнения SQL

🪙 Аналогия копилки с монетами
1

WHERE — убираем гнутые и иностранные монеты до раскладки.

2

GROUP BY — раскладываем монеты в стопки (10, 50, 100 ₸).

3

Агрегаты — считаем COUNT, SUM, AVG, MIN, MAX в стопках.

4

HAVING — оставляем стопки, где набралось больше 5 монет.

📌 3 следствия из конвейера
  • WHERE не видит агрегатов (COUNT, SUM) — групп ещё нет!
  • HAVING не видит псевдонимов AS из SELECT — пиши формулу целиком.
  • Любая колонка в SELECT обязана быть либо в GROUP BY, либо внутри агрегата.
⚡
Всё, что можно отсечь до группировки (например, status = 'paid'), режь в WHERE — так GROUP BY переварит меньше строк!

Клиентский отчёт лояльности: условная агрегация

SQL 
SELECT
    c.full_name,
    COUNT(DISTINCT CASE WHEN o.status = 'paid' THEN o.id END)      AS paid_orders,
    COUNT(DISTINCT CASE WHEN o.status = 'cancelled' THEN o.id END) AS cancelled_orders,
    COALESCE(SUM(CASE WHEN o.status = 'paid' THEN oi.quantity * p.price ELSE 0.00 END), 0.00) AS total_spent,
    CASE
        WHEN COALESCE(SUM(CASE WHEN o.status = 'paid' THEN oi.quantity * p.price ELSE 0.00 END), 0.00) >= 30000 THEN 'VIP'
        WHEN COALESCE(SUM(CASE WHEN o.status = 'paid' THEN oi.quantity * p.price ELSE 0.00 END), 0.00) > 0      THEN 'Активный'
        ELSE 'Спящий'
    END AS segment
FROM customers AS c
LEFT JOIN orders AS o      ON o.customer_id = c.id
LEFT JOIN order_items AS oi ON oi.order_id = o.id
LEFT JOIN products AS p     ON p.id = oi.product_id
GROUP BY c.id, c.full_name
ORDER BY total_spent DESC, c.id;
-- Официальный стандарт SQL в PostgreSQL: FILTER (WHERE ...) читается намного чище!
SELECT
    c.full_name,
    COUNT(DISTINCT o.id) FILTER (WHERE o.status = 'paid')      AS paid_orders,
    COUNT(DISTINCT o.id) FILTER (WHERE o.status = 'cancelled') AS cancelled_orders,
    COALESCE(SUM(oi.quantity * p.price) FILTER (WHERE o.status = 'paid'), 0.00) AS total_spent
FROM customers AS c
LEFT JOIN orders AS o      ON o.customer_id = c.id
LEFT JOIN order_items AS oi ON oi.order_id = o.id
LEFT JOIN products AS p     ON p.id = oi.product_id
GROUP BY c.id, c.full_name
ORDER BY total_spent DESC, c.id;

✗ Как НЕ надо (NOT IN + NULL = 0 строк!)
SELECT title FROM products
WHERE category NOT IN (SELECT city FROM customers);

03 · Обобщённые табличные выражения (CTE)

WITH: раскладываем сложный запрос по полочкам

SQL 
WITH customer_totals AS (
  SELECT
    c.id AS customer_id,
    INITCAP(TRIM(c.full_name)) AS customer_name,
    COUNT(DISTINCT o.id) AS paid_orders,
    SUM(oi.quantity * oi.unit_price) AS total_spent
  FROM customers c
  JOIN orders o ON o.customer_id = c.id
  JOIN order_items oi ON oi.order_id = o.id
  WHERE o.status = 'paid'
  GROUP BY c.id, c.full_name
),
store_avg AS (
  SELECT ROUND(AVG(total_spent), 2) AS avg_spent FROM customer_totals
)
-- Продолжение запроса после WITH: переиспользуем оба шага без дублирования!
SELECT
  ct.customer_name,
  ct.paid_orders,
  ct.total_spent,
  sa.avg_spent,
  ct.total_spent - sa.avg_spent AS above_avg
FROM customer_totals ct
CROSS JOIN store_avg sa
WHERE ct.total_spent > sa.avg_spent
ORDER BY ct.total_spent DESC;

03 · Читаемость и планировщик

CTE против «матрёшки» подзапросов и MATERIALIZED

✗ «Матрёшка» с дублированием логики
SELECT t.customer_name, t.total_spent
FROM (
  SELECT c.full_name AS customer_name,
         SUM(oi.quantity * oi.unit_price) AS total_spent
  FROM customers c JOIN orders o ...
  WHERE o.status = 'paid' GROUP BY c.id, c.full_name
) t
WHERE t.total_spent > (
  SELECT AVG(sub.s) FROM (
    SELECT SUM(oi.quantity * oi.unit_price) AS s
    FROM orders o JOIN order_items oi ...
    WHERE o.status = 'paid' GROUP BY o.customer_id
  ) sub
);

Расчёт дублируется дважды: при смене условия оплаты легко поправить одно место и забыть второе.

✓ Один источник правды в CTE
WITH customer_totals AS (
  SELECT c.full_name AS customer_name,
         SUM(oi.quantity * oi.unit_price) AS total_spent
  FROM customers c
  JOIN orders o ON o.customer_id = c.id
  JOIN order_items oi ON oi.order_id = o.id
  WHERE o.status = 'paid'
  GROUP BY c.id, c.full_name
)
SELECT customer_name, total_spent
FROM customer_totals
WHERE total_spent > (
  SELECT AVG(total_spent) FROM customer_totals
);

Читается сверху вниз как книга, а любой шаг легко отладить отдельно через SELECT * FROM customer_totals;.

⚡
PostgreSQL 12+: планировщик сам решает, встроить ли CTE в запрос или сохранить в память. При необходимости ты можешь управлять этим явно: WITH step AS MATERIALIZED (...) или AS NOT MATERIALIZED (...).

02 · Суть оконных функций

GROUP BY против OVER (PARTITION BY ...)

Часть синтаксиса Зачем нужна (аналогия) Обязательно?
функция(...) Что считаем: AVG, SUM, COUNT, ROW_NUMBER, LAG. Да
OVER () Пустые скобки = «смотри сразу на всю таблицу целиком». Да
PARTITION BY col «Подели таблицу на комнаты» (категории, клиенты) и считай в каждой отдельно. Нет
ORDER BY col Порядок строк внутри комнаты (нужен для рейтингов, LAG, накопительных сумм). По задаче
ROWS BETWEEN ... Подвижная рамка: сколько строк до и после текущей брать в расчёт. Нет

WITH monthly_sales AS (

         SUM(oi.quantity * oi.unit_price) AS order_total
  FROM orders o JOIN order_items oi ON oi.order_id = o.id
  WHERE o.status = 'paid' GROUP BY o.id, o.created_at
)

  SUM(order_total) OVER (ORDER BY order_date, order_id
    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running_total,
  ROUND(AVG(order_total) OVER (ORDER BY order_date, order_id
    ROWS BETWEEN 2 PRECEDING AND CURRENT ROW), 2) AS moving_avg_3
FROM order_totals ORDER BY order_date, order_id;

05 · Ловушка рамок и суперсила OVER ()

Ловушка RANGE при равных датах и доли через OVER ()

✗ Как НЕ надо (скрытый режим RANGE)
SUM(order_total) OVER (
  ORDER BY order_date
)
  • Без явного ROWS PostgreSQL включает режим RANGE.
  • Если в один день пришло 2 заказа, они сложатся вместе сразу в обеих строках, и накопительный итог «прыгнет»!
✓ Как правильно (явный ROWS BETWEEN + id)
SUM(order_total) OVER (
  ORDER BY order_date, order_id
  ROWS BETWEEN UNBOUNDED PRECEDING
           AND CURRENT ROW
)
  • Явный ROWS BETWEEN и уникальный order_id гарантируют честное пошаговое накопление строка за строкой.
🥧
Суперсила пустого OVER () для процентов: чтобы узнать долю каждой строки в общей сумме таблицы, напиши ROUND(order_total * 100.0 / SUM(order_total) OVER (), 1) AS share_pct!

Как меняется чек клиента от покупки к покупке?

SQL Эталонное решение · Задача 3 
WITH order_totals AS (
  SELECT o.customer_id, c.full_name, o.id AS order_id, o.created_at::date AS order_date,
         SUM(oi.quantity * oi.unit_price) AS order_total
  FROM orders o

  JOIN order_items oi ON oi.order_id = o.id
  WHERE o.status = 'paid'
  GROUP BY o.customer_id, c.full_name, o.id, o.created_at
)
SELECT full_name, order_date, order_total,
       LAG(order_total) OVER w               AS prev_total,
       order_total - LAG(order_total) OVER w AS diff_total
FROM order_totals

ORDER BY customer_id, order_date, order_id;

SQL Эталонное решение · Задача 4 (2 CTE + 3 оконные функции) 
WITH order_totals AS (
  SELECT o.customer_id, INITCAP(TRIM(c.full_name)) AS customer_name,
         o.id AS order_id, o.created_at::date AS order_date,
         SUM(oi.quantity * oi.unit_price) AS order_total
  FROM orders o JOIN customers c ON c.id = o.customer_id
  JOIN order_items oi ON oi.order_id = o.id WHERE o.status = 'paid'
  GROUP BY o.customer_id, c.full_name, o.id, o.created_at
),
enriched AS (
  SELECT *,
    COUNT(*) OVER (PARTITION BY customer_id) AS total_orders,
    ROW_NUMBER() OVER (PARTITION BY customer_id ORDER BY order_date, order_id) AS purchase_num,
    order_total - LAG(order_total) OVER (PARTITION BY customer_id ORDER BY order_date, order_id) AS diff_from_prev
  FROM order_totals
)
SELECT customer_name, purchase_num, order_date, order_total, diff_from_prev
FROM enriched WHERE total_orders >= 2 ORDER BY customer_name, purchase_num;
