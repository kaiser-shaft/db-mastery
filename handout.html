UPDATE, DELETE, Soft Delete и INNER JOINУрок 1.4

-- Точечное обновление с возвратом изменённой строки
UPDATE products
SET stock = stock + 15, price = 4200.00
WHERE id = 6
RETURNING id, title, stock, price;

-- Физическое удаление vs Мягкое удаление (Soft Delete)
DELETE FROM products WHERE id = 9 RETURNING *;
UPDATE products SET is_active = false WHERE id = 4 RETURNING id, is_active;

-- Внутреннее соединение 3 таблиц (INNER JOIN)
SELECT
    o.id         AS order_id,
    c.name       AS category_name,
    p.title      AS product_title,
    o.quantity,
    (p.price * o.quantity) AS total_cost
FROM orders AS o
INNER JOIN products AS p   ON o.product_id = p.id
INNER JOIN categories AS c ON p.category_id = c.id
WHERE o.status = 'paid'
ORDER BY total_cost DESC;

JOIN-ы, UNION, CASE, GROUP BY и HAVINGУрок 2.2

Порядок: 1.FROM/JOIN → 2.WHERE → 3.GROUP BY → 4.HAVING → 5.SELECT → 6.ORDER BY
-- LEFT JOIN: фильтр правой таблицы строго в ON (иначе станет INNER JOIN!)
SELECT c.name, COUNT(o.id) AS paid_cnt
FROM customers c
LEFT JOIN orders o ON c.id = o.customer_id AND o.status = 'paid'
GROUP BY c.id, c.name;

-- Anti-Join: поиск клиентов без единого заказа
SELECT c.id, c.name FROM customers c
LEFT JOIN orders o ON c.id = o.customer_id
WHERE o.id IS NULL;

-- Условная агрегация: COUNT(o.id), НЕ COUNT(*) после LEFT JOIN!
SELECT
    c.city,
    COUNT(o.id) AS total_orders,
    COUNT(o.id) FILTER (WHERE o.status = 'paid') AS paid_orders,
    COALESCE(SUM(CASE WHEN o.status = 'paid' THEN o.amount END), 0) AS revenue
FROM customers c
LEFT JOIN orders o ON c.id = o.customer_id
GROUP BY c.city                      -- В SELECT только поля GROUP BY или агрегаты
HAVING SUM(o.amount) FILTER (WHERE o.status = 'paid') >= 10000;

-- UNION (удаляет дубли) vs UNION ALL (быстро, без дедупликации)
SELECT email FROM customers UNION ALL SELECT email FROM leads;

Подзапросы, CTE, Даты, Строки и JSONBУрок 2.3

Оператор JSONBТип возврата / Назначение

-- NOT EXISTS безопасен при NULL (в отличие от NOT IN)
SELECT p.id, p.title FROM products p
WHERE NOT EXISTS (SELECT 1 FROM order_items oi WHERE oi.product_id = p.id);

-- CTE, работа с датами и строками
WITH monthly AS (
    SELECT
        DATE_TRUNC('month', created_at)::date AS m,
        EXTRACT(YEAR FROM created_at) AS yr,
        SPLIT_PART(LOWER(TRIM(email)), '@', 2) AS domain,
        LPAD(id::text, 5, '0') AS code
    FROM orders
    WHERE created_at >= '2026-01-01' AND created_at < '2027-01-01'
      AND created_at >= now() - INTERVAL '3 months'
)
SELECT domain, COUNT(*) FROM monthly GROUP BY domain;

-- Разворот массива JSONB и проверка вхождения
SELECT p.title, tag
FROM products p, jsonb_array_elements_text(p.specs -> 'tags') AS tag
WHERE p.specs @> '{"in_stock": true}'::jsonb;

Оконные функции и аналитические паттерныУроки 2.4 – 2.5

По умолч. с ORDER BY действует RANGE (объединяет дубли); для строк пишите ROWS BETWEEN
ФункцияНазначение / Ранжирование дублей
ROW_NUMBER()Уникальные номера: 1, 2, 3, 4 (нужен тай-брейкер id)
RANK()Ранг с пропусками: 1, 2, 2, 4
DENSE_RANK()Ранг без пропусков: 1, 2, 2, 3
LAG(c,1,0) / LEAD(c,1,0)Предыдущая / следующая строка без self-join
 
-- Топ-N в каждой категории (фильтрация окна только через CTE/подзапрос)
WITH ranked AS (
    SELECT category_id, title, price,
           ROW_NUMBER() OVER (PARTITION BY category_id ORDER BY price DESC, id) AS rn
    FROM products
)
SELECT * FROM ranked WHERE rn <= 3;

-- Накопительный итог, скользящее среднее, MoM% и доля от общего
SELECT
    dt, revenue,
    LAG(revenue, 1) OVER w AS prev_rev,
    ROUND((revenue - LAG(revenue, 1) OVER w) * 100.0
          / NULLIF(LAG(revenue, 1) OVER w, 0), 1) AS mom_pct,
    SUM(revenue) OVER (w ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS run_total,
    ROUND(AVG(revenue) OVER (w ROWS BETWEEN 2 PRECEDING AND CURRENT ROW), 2) AS ma_3,
    ROUND(revenue * 100.0 / NULLIF(SUM(revenue) OVER (), 0), 2) AS share_pct
FROM daily_sales
WINDOW w AS (PARTITION BY store_id ORDER BY dt);

Сквозной контур: OLTP (PostgreSQL) → ELT → OLAP (ClickHouse: REPLACE PARTITION & DQ)Урок 6.5

-- 1. Идемпотентный ELT (Delete-Insert по дате в PG)
CREATE OR REPLACE PROCEDURE dwh.run_elt(p_dt DATE) LANGUAGE plpgsql AS $$
BEGIN
    DELETE FROM dwh.fact_sales WHERE sale_date = p_dt;
    INSERT INTO dwh.fact_sales (sale_date, product_sk, amount)
    SELECT s.sale_date, d.product_sk, s.amount
    FROM stg.sales s JOIN dwh.dim_product d ON d.product_bk = s.product_id
    WHERE s.sale_date = p_dt;
END; $$;
CALL dwh.run_elt(DATE '2026-02-01');

-- 4. Сверка качества данных (DQ Reconciliation: PG vs CH)
WITH
  pg AS (SELECT count() AS r, sum(amount) AS s
         FROM postgresql('pg_host:5432', 'db', 'export_mart', 'user', 'pass', 'dwh')),
  ch AS (SELECT count() AS r, sum(amount) AS s FROM mart_sales)
SELECT pg.r AS pg_rows, ch.r AS ch_rows, pg.s AS pg_sum, ch.s AS ch_sum,
       if(pg.r = ch.r AND pg.s = ch.s, 'OK', 'MISMATCH') AS dq_status
FROM pg CROSS JOIN ch;
